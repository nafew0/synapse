  \question[10]
  Find the derivative of $f(x) = x^3 \sin x$.
  \begin{solution}
    By the product rule,
    \[
      f'(x) = 3x^2 \sin x + x^3 \cos x.
    \]
  \end{solution}

  \question
  Let $A = \begin{pmatrix} 1 & 2 \\ 3 & 4 \end{pmatrix}$.
  \begin{parts}
    \part[4] Find $\det A$.
    \begin{solution}
      $\det A = 1 \cdot 4 - 2 \cdot 3 = -2$.
    \end{solution}
    \part[6] Find $A^{-1}$.
    \begin{solution}
      \[
        A^{-1} = \frac{1}{\det A}
                 \begin{pmatrix} 4 & -2 \\ -3 & 1 \end{pmatrix}
               = \begin{pmatrix} -2 & 1 \\ 3/2 & -1/2 \end{pmatrix}.
      \]
    \end{solution}
  \end{parts}

  \question[5]
  State the trapezoidal rule for $\int_a^b f(x)\,dx$ with one interval.
  \begin{solution}
    \[
      \int_a^b f(x)\,dx \approx \frac{b - a}{2}\bigl(f(a) + f(b)\bigr).
    \]
  \end{solution}

\end{questions}
