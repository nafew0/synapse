\section{Method}
\label{sec:method}

The model is
\begin{equation}
  y_i = \beta_0 + \beta_1 x_i + \varepsilon_i,
  \qquad \varepsilon_i \sim \mathcal{N}(0, \sigma^2),
  \label{eq:model}
\end{equation}
and the parameters in~\eqref{eq:model} are estimated by least squares:
\begin{align}
  \hat{\beta}_1 &= \frac{\sum_{i=1}^{n} (x_i - \bar{x})(y_i - \bar{y})}
                        {\sum_{i=1}^{n} (x_i - \bar{x})^2}, \label{eq:slope} \\
  \hat{\beta}_0 &= \bar{y} - \hat{\beta}_1 \bar{x}. \label{eq:intercept}
\end{align}
